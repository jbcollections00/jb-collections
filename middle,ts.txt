import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const { pathname, hostname } = request.nextUrl;

  // 1. LOCALHOST / DEVELOPMENT BYPASS
  // Kapag nagte-test ka sa sarili mong computer (npm run dev), hind gagana ang Maintenance block.
  // Makakapag-edit at makakapag-preview ka sa lahat ng sub-pages nang walang problema.
  if (
    process.env.NODE_ENV === 'development' || 
    hostname === 'localhost' || 
    hostname === '127.0.0.1'
  ) {
    return NextResponse.next();
  }

  // --- PRODUCTION MAINTENANCE MODE (Para sa mga Bisita sa Live Site) ---

  // 2. Payagan ang Next.js system files, CSS, JavaScript, at static images
  if (
    pathname.startsWith('/_next') ||
    pathname.startsWith('/static') ||
    pathname.includes('.')
  ) {
    return NextResponse.next();
  }

  // 3. Kung nasa homepage ('/') na ang bisita, ipakita ang maintenance page
  if (pathname === '/') {
    return NextResponse.next();
  }

  // 4. I-rewrite ang lahat ng sub-pages (/gallery, /admin, etc.) pabalik sa '/'
  return NextResponse.rewrite(new URL('/', request.url));
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
};
